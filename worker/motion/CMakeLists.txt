cmake_minimum_required(VERSION 3.24)
project(scatter_motion LANGUAGES CXX)
option(SCATTER_LIVE_BUILD "Interactive actual-step latest frame/pacing" OFF)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Pinned MSVC x64 required")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BUILD_SUFFIX motion)
if(SCATTER_LIVE_BUILD)
  set(BUILD_SUFFIX live)
endif()
if(NOT CMAKE_BINARY_DIR STREQUAL "${ROOT}/build-review-${BUILD_SUFFIX}" AND NOT (SCATTER_LIVE_BUILD AND (CMAKE_BINARY_DIR STREQUAL "${ROOT}/build-review-live-io1" OR CMAKE_BINARY_DIR STREQUAL "${ROOT}/build-review-live-ready1")))
  message(FATAL_ERROR "Use the isolated build-review-motion or build-review-live directory")
endif()
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CXX_STANDARD 17)
set(DEPS "${ROOT}/build/_deps")
set(JOLT_LIB "${ROOT}/build-review-pbr/_deps/jolt-build/Release/Jolt.lib")
set(ZLIB_LIB "${ROOT}/build-review-pbr/_deps/zlib-build/Release/zs.lib")
foreach(LIB IN ITEMS "${JOLT_LIB}" "${ZLIB_LIB}")
  if(NOT EXISTS "${LIB}")
    message(FATAL_ERROR "Missing ${LIB}; build the review-pbr dependencies first")
  endif()
endforeach()
add_library(PinnedJolt STATIC IMPORTED)
set_target_properties(PinnedJolt PROPERTIES IMPORTED_LOCATION "${JOLT_LIB}")
add_library(PinnedZlib STATIC IMPORTED)
set_target_properties(PinnedZlib PROPERTIES IMPORTED_LOCATION "${ZLIB_LIB}")
set(INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${JOLT_LIB}" "${ZLIB_LIB}")
foreach(DIR IN ITEMS "${ROOT}/worker" "${ROOT}/experiments/periodic_solver" "${ROOT}/experiments/periodic_backend" "${DEPS}/jolt-src/Jolt" "${DEPS}/json-src/include" "${DEPS}/tinyobj-src" "${DEPS}/lodepng-src" "${DEPS}/zlib-src")
  file(GLOB_RECURSE DEP CONFIGURE_DEPENDS "${DIR}/*.h" "${DIR}/*.hpp" "${DIR}/*.cpp" "${DIR}/*.c")
  list(APPEND INPUTS ${DEP})
endforeach()
list(REMOVE_DUPLICATES INPUTS)
list(SORT INPUTS)
set(MATERIAL "motion=0.4.0-motion.1\nlive=${SCATTER_LIVE_BUILD}\nprotocol=0.3.0\ncompiler=${CMAKE_CXX_COMPILER_ID}/${CMAKE_CXX_COMPILER_VERSION}\nmain=/MT /O2 /fp:precise\nsolver=/MT /O2 /fp:fast /fp:except-\n")
foreach(INPUT IN LISTS INPUTS)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${INPUT}")
  file(SHA256 "${INPUT}" HASH)
  file(RELATIVE_PATH REL "${ROOT}" "${INPUT}")
  string(APPEND MATERIAL "${REL}=${HASH}\n")
endforeach()
foreach(MODE IN ITEMS legacy pbr periodic)
  set(TARGET "scatter_worker_${MODE}_${BUILD_SUFFIX}")
  set(SOURCES "${ROOT}/worker/main.cpp" "${DEPS}/lodepng-src/lodepng.cpp")
  if(MODE STREQUAL periodic)
    list(APPEND SOURCES "${ROOT}/worker/periodic/solver.cpp")
    set_source_files_properties("${ROOT}/worker/periodic/solver.cpp" PROPERTIES COMPILE_OPTIONS "/fp:fast;/fp:except-")
  endif()
  add_executable(${TARGET} ${SOURCES})
  target_include_directories(${TARGET} PRIVATE "${DEPS}/jolt-src" "${DEPS}/json-src/include" "${DEPS}/tinyobj-src" "${DEPS}/lodepng-src" "${DEPS}/zlib-src" "${ROOT}/build-review-pbr/_deps/zlib-build")
  target_compile_definitions(${TARGET} PRIVATE NDEBUG _MBCS WIN32 _WINDOWS NOMINMAX JPH_FLOATING_POINT_EXCEPTIONS_ENABLED JPH_DEBUG_RENDERER JPH_PROFILE_ENABLED JPH_OBJECT_STREAM JPH_USE_SSE4_1 JPH_USE_SSE4_2 SCATTER_MOTION_BUILD=1)
  if(SCATTER_LIVE_BUILD)
    target_compile_definitions(${TARGET} PRIVATE SCATTER_LIVE_BUILD=1)
  endif()
  if(NOT MODE STREQUAL legacy)
    target_compile_definitions(${TARGET} PRIVATE SCATTER_REVIEW_BUILD=1 SCATTER_REVIEW_PBR_BUILD=1)
  endif()
  if(MODE STREQUAL periodic)
    target_compile_definitions(${TARGET} PRIVATE SCATTER_REVIEW_PERIODIC_BUILD=1)
  endif()
  file(WRITE "${CMAKE_BINARY_DIR}/${MODE}-build-inputs.txt" "${MATERIAL}mode=${MODE}\n")
  file(SHA256 "${CMAKE_BINARY_DIR}/${MODE}-build-inputs.txt" FINGERPRINT)
  file(WRITE "${CMAKE_BINARY_DIR}/${MODE}-build-fingerprint.txt" "${FINGERPRINT}\n")
  target_compile_definitions(${TARGET} PRIVATE SCATTER_MOTION_BUILD_FINGERPRINT="${FINGERPRINT}" SCATTER_REVIEW_BUILD_FINGERPRINT="${FINGERPRINT}")
  target_compile_options(${TARGET} PRIVATE /W4 /utf-8 /O2 /Ob2 /fp:precise /GR /EHsc /Zc:__cplusplus)
  target_link_libraries(${TARGET} PRIVATE PinnedJolt PinnedZlib bcrypt)
  set_target_properties(${TARGET} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
