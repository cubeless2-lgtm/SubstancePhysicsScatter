cmake_minimum_required(VERSION 3.24)
project(scatter_periodic_review LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Pinned MSVC x64 required")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT CMAKE_BINARY_DIR STREQUAL "${ROOT}/build-review-periodic")
  message(FATAL_ERROR "Use the isolated build-review-periodic directory")
endif()
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CXX_STANDARD 17)
set(JOLT_LIB "${ROOT}/build-review-pbr/_deps/jolt-build/Release/Jolt.lib")
set(ZLIB_LIB "${ROOT}/build-review-pbr/_deps/zlib-build/Release/zs.lib")
foreach(LIB IN ITEMS "${JOLT_LIB}" "${ZLIB_LIB}")
  if(NOT EXISTS "${LIB}")
    message(FATAL_ERROR "Missing ${LIB}; build the review-pbr dependencies first")
  endif()
endforeach()
set(DEPS "${ROOT}/build/_deps")
add_library(PinnedJolt STATIC IMPORTED)
set_target_properties(PinnedJolt PROPERTIES IMPORTED_LOCATION "${JOLT_LIB}")
add_library(PinnedZlib STATIC IMPORTED)
set_target_properties(PinnedZlib PROPERTIES IMPORTED_LOCATION "${ZLIB_LIB}")
add_executable(scatter_worker_review ../main.cpp solver.cpp "${DEPS}/lodepng-src/lodepng.cpp")
target_include_directories(scatter_worker_review PRIVATE "${DEPS}/jolt-src" "${DEPS}/json-src/include" "${DEPS}/tinyobj-src" "${DEPS}/lodepng-src" "${DEPS}/zlib-src" "${ROOT}/build-review-pbr/_deps/zlib-build")
target_compile_definitions(scatter_worker_review PRIVATE NDEBUG _MBCS WIN32 _WINDOWS NOMINMAX JPH_FLOATING_POINT_EXCEPTIONS_ENABLED JPH_DEBUG_RENDERER JPH_PROFILE_ENABLED JPH_OBJECT_STREAM JPH_USE_SSE4_1 JPH_USE_SSE4_2 SCATTER_REVIEW_BUILD=1 SCATTER_REVIEW_PBR_BUILD=1 SCATTER_REVIEW_PERIODIC_BUILD=1)
target_compile_options(scatter_worker_review PRIVATE /W4 /utf-8 /O2 /Ob2 /fp:precise /GR /EHsc /Zc:__cplusplus)
set_source_files_properties(solver.cpp PROPERTIES COMPILE_OPTIONS "/fp:fast;/fp:except-")
target_link_libraries(scatter_worker_review PRIVATE PinnedJolt PinnedZlib bcrypt)
set_target_properties(scatter_worker_review PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
# Hash only actual translation-unit dependencies and cached dependencies. No runtime/product inventory.
set(MATERIAL "implementation=0.4.0-review.3\nprotocol=0.3.0\ncompiler=${CMAKE_CXX_COMPILER_ID}/${CMAKE_CXX_COMPILER_VERSION}\ntoolset=${MSVC_TOOLSET_VERSION}/${CMAKE_VS_PLATFORM_TOOLSET_VERSION}\nmain=/MT /O2 /fp:precise\nsolver=/MT /O2 /fp:fast /fp:except-\n")
set(INPUTS "${ROOT}/worker/main.cpp" "${ROOT}/worker/compatibility.h" "${ROOT}/worker/mask.h" "${ROOT}/worker/review_hulls.h" "${ROOT}/worker/review_pbr_input.h" "${ROOT}/worker/review_pbr_bake.h" "${ROOT}/experiments/periodic_solver/kernel.h" "${ROOT}/experiments/periodic_backend/geometry.h" "${JOLT_LIB}" "${ZLIB_LIB}" "${ROOT}/third_party/dependencies.json")
file(GLOB LOCAL CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(APPEND INPUTS ${LOCAL})
foreach(DIR IN ITEMS "${DEPS}/jolt-src/Jolt" "${DEPS}/json-src/include" "${DEPS}/tinyobj-src" "${DEPS}/lodepng-src" "${DEPS}/zlib-src")
  file(GLOB_RECURSE DEP CONFIGURE_DEPENDS "${DIR}/*.h" "${DIR}/*.hpp" "${DIR}/*.cpp" "${DIR}/*.c")
  list(APPEND INPUTS ${DEP})
endforeach()
list(REMOVE_DUPLICATES INPUTS)
list(SORT INPUTS)
foreach(INPUT IN LISTS INPUTS)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${INPUT}")
  file(SHA256 "${INPUT}" HASH)
  file(RELATIVE_PATH REL "${ROOT}" "${INPUT}")
  string(APPEND MATERIAL "${REL}=${HASH}\n")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/review-build-inputs.txt" "${MATERIAL}")
file(SHA256 "${CMAKE_BINARY_DIR}/review-build-inputs.txt" FINGERPRINT)
file(WRITE "${CMAKE_BINARY_DIR}/review-build-fingerprint.txt" "${FINGERPRINT}\n")
target_compile_definitions(scatter_worker_review PRIVATE SCATTER_REVIEW_BUILD_FINGERPRINT="${FINGERPRINT}")
if(EXISTS "${ROOT}/tests/periodic_bake_probe.cpp")
  add_executable(scatter_periodic_bake_probe EXCLUDE_FROM_ALL "${ROOT}/tests/periodic_bake_probe.cpp" solver.cpp "${DEPS}/lodepng-src/lodepng.cpp")
  foreach(PROP IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(VALUE scatter_worker_review ${PROP})
    set_target_properties(scatter_periodic_bake_probe PROPERTIES ${PROP} "${VALUE}")
  endforeach()
  target_link_libraries(scatter_periodic_bake_probe PRIVATE PinnedJolt PinnedZlib bcrypt)
  set_target_properties(scatter_periodic_bake_probe PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
endif()
