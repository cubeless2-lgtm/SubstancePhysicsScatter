cmake_minimum_required(VERSION 3.24)
project(SubstancePhysicsScatter VERSION 0.3.0 LANGUAGES C CXX)
option(SCATTER_REVIEW_BUILD "Build isolated opt-in review worker; legacy wire remains 0.3.0" OFF)
option(SCATTER_REVIEW_PBR_BUILD "Enable material review in a separate review-pbr build" OFF)
if(SCATTER_REVIEW_PBR_BUILD)
  if(NOT SCATTER_REVIEW_BUILD OR CMAKE_BINARY_DIR STREQUAL "${CMAKE_SOURCE_DIR}/build" OR CMAKE_BINARY_DIR STREQUAL "${CMAKE_SOURCE_DIR}/build-review")
    message(FATAL_ERROR "PBR review requires SCATTER_REVIEW_BUILD and its own build-review-pbr directory")
  endif()
endif()
if(SCATTER_REVIEW_BUILD AND CMAKE_BINARY_DIR STREQUAL "${CMAKE_SOURCE_DIR}/build")
  message(FATAL_ERROR "Review build must not overwrite the accepted legacy build directory")
endif()
if(SCATTER_REVIEW_BUILD)
  set(CMAKE_CONFIGURATION_TYPES "Release" CACHE STRING "Review fingerprint is Release-only" FORCE)
  set(CMAKE_BUILD_TYPE "Release" CACHE STRING "Review fingerprint is Release-only" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(OVERRIDE_CXX_FLAGS OFF CACHE BOOL "" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
set(CPP_RTTI_ENABLED ON CACHE BOOL "" FORCE)
set(CPP_EXCEPTIONS_ENABLED ON CACHE BOOL "" FORCE)
set(USE_AVX OFF CACHE BOOL "" FORCE)
set(USE_AVX2 OFF CACHE BOOL "" FORCE)
set(USE_F16C OFF CACHE BOOL "" FORCE)
set(USE_FMADD OFF CACHE BOOL "" FORCE)
set(USE_LZCNT OFF CACHE BOOL "" FORCE)
set(USE_TZCNT OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(jolt URL https://github.com/jrouwe/JoltPhysics/archive/a63aa3b8e24cf95f3fab2613f9a3015b164ef62c.zip SOURCE_SUBDIR Build)
FetchContent_Declare(json URL https://github.com/nlohmann/json/archive/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03.zip)
FetchContent_Declare(tinyobj URL https://github.com/tinyobjloader/tinyobjloader/archive/2945a967c5303b2c8c14174117c45f3302591150.zip)
FetchContent_Declare(lodepng URL https://github.com/lvandeve/lodepng/archive/e584a8490db1b76b81bf298c11c09d3b55d9c270.zip)
FetchContent_MakeAvailable(jolt json tinyobj lodepng)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib URL https://www.zlib.net/zlib-1.3.2.tar.gz URL_HASH SHA256=bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16 DOWNLOAD_DIR ${CMAKE_BINARY_DIR}/downloads)
FetchContent_MakeAvailable(zlib)
add_executable(scatter_worker worker/main.cpp ${lodepng_SOURCE_DIR}/lodepng.cpp)
if(SCATTER_REVIEW_BUILD)
  set(REVIEW_IMPLEMENTATION "0.4.0-review.1")
  if(SCATTER_REVIEW_PBR_BUILD)
    set(REVIEW_IMPLEMENTATION "0.4.0-review.2")
    target_compile_definitions(scatter_worker PRIVATE SCATTER_REVIEW_PBR_BUILD=1)
  endif()
  set(REVIEW_BUILD_MATERIAL "implementation=${REVIEW_IMPLEMENTATION}\nprotocol=0.3.0\ncompiler=${CMAKE_CXX_COMPILER_ID}/${CMAKE_CXX_COMPILER_VERSION}\ngenerator=${CMAKE_GENERATOR}/${CMAKE_GENERATOR_PLATFORM}\nflags=${CMAKE_CXX_FLAGS}/${CMAKE_CXX_FLAGS_RELEASE}\ncrt=${CMAKE_MSVC_RUNTIME_LIBRARY}\nconfiguration=${CMAKE_BUILD_TYPE}\n")
  foreach(REVIEW_ROOT IN ITEMS "${CMAKE_SOURCE_DIR}/worker" "${jolt_SOURCE_DIR}/Jolt" "${json_SOURCE_DIR}/include" "${tinyobj_SOURCE_DIR}" "${lodepng_SOURCE_DIR}" "${zlib_SOURCE_DIR}")
    file(GLOB_RECURSE REVIEW_INPUTS CONFIGURE_DEPENDS "${REVIEW_ROOT}/*.h" "${REVIEW_ROOT}/*.hpp" "${REVIEW_ROOT}/*.cpp" "${REVIEW_ROOT}/*.c" "${REVIEW_ROOT}/*.cmake" "${REVIEW_ROOT}/CMakeLists.txt")
    list(SORT REVIEW_INPUTS)
    get_filename_component(REVIEW_ROLE "${REVIEW_ROOT}" NAME)
    foreach(REVIEW_INPUT IN LISTS REVIEW_INPUTS)
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REVIEW_INPUT}")
      file(SHA256 "${REVIEW_INPUT}" REVIEW_HASH)
      file(RELATIVE_PATH REVIEW_REL "${REVIEW_ROOT}" "${REVIEW_INPUT}")
      string(APPEND REVIEW_BUILD_MATERIAL "${REVIEW_ROLE}/${REVIEW_REL}=${REVIEW_HASH}\n")
    endforeach()
  endforeach()
  file(SHA256 "${CMAKE_SOURCE_DIR}/CMakeLists.txt" REVIEW_CMAKE_HASH)
  file(SHA256 "${CMAKE_SOURCE_DIR}/third_party/dependencies.json" REVIEW_PINS_HASH)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/third_party/dependencies.json")
  string(APPEND REVIEW_BUILD_MATERIAL "CMakeLists=${REVIEW_CMAKE_HASH}\ndependency-lock=${REVIEW_PINS_HASH}\n")
  file(WRITE "${CMAKE_BINARY_DIR}/review-build-inputs.txt" "${REVIEW_BUILD_MATERIAL}")
  file(SHA256 "${CMAKE_BINARY_DIR}/review-build-inputs.txt" REVIEW_BUILD_FINGERPRINT)
  file(WRITE "${CMAKE_BINARY_DIR}/review-build-fingerprint.txt" "${REVIEW_BUILD_FINGERPRINT}\n")
  target_compile_definitions(scatter_worker PRIVATE SCATTER_REVIEW_BUILD=1 SCATTER_REVIEW_BUILD_FINGERPRINT="${REVIEW_BUILD_FINGERPRINT}")
  set_target_properties(scatter_worker PROPERTIES OUTPUT_NAME scatter_worker_review)
endif()
target_include_directories(scatter_worker PRIVATE ${tinyobj_SOURCE_DIR} ${lodepng_SOURCE_DIR})
target_link_libraries(scatter_worker PRIVATE Jolt nlohmann_json::nlohmann_json)
target_link_libraries(scatter_worker PRIVATE bcrypt ZLIB::ZLIBSTATIC)
if(MSVC)
  target_compile_options(scatter_worker PRIVATE /W4 /utf-8)
endif()
install(TARGETS scatter_worker RUNTIME DESTINATION bin)
